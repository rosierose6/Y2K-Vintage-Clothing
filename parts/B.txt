  // Mobile slide-in menu wiring
  const menuBtn = document.querySelector('.menu-btn');
  const mobileNav = document.getElementById('mobileNav');
  const setNav = (open) => {
    mobileNav.classList.toggle('open', open);
    mobileNav.setAttribute('aria-hidden', open ? 'false' : 'true');
    document.body.style.overflow = open ? 'hidden' : '';
  };
  if (menuBtn && mobileNav) {
    menuBtn.addEventListener('click', () => setNav(true));
    mobileNav.querySelectorAll('[data-close]').forEach(el => el.addEventListener('click', () => setNav(false)));
    addEventListener('keydown', (e) => { if (e.key === 'Escape') setNav(false); });
  }

  // Gold fairy dust: slow rise-and-fall the length of the page, layered with scroll parallax
  const dustLayers = document.querySelectorAll('.sparkle, .hero-dust');
  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const dustPos = (y, d) => `0 ${(-y*0.5 - d).toFixed(1)}px, 40px ${(50 - y*0.72 - d*0.65).toFixed(1)}px, 80px ${(24 - y*0.32 - d*1.35).toFixed(1)}px`;

  if (reduceMotion) {
    const onScroll = () => dustLayers.forEach(el => { el.style.backgroundPosition = dustPos(window.scrollY || 0, 0); });
    addEventListener('scroll', onScroll, { passive: true });
    onScroll();
  } else {
    const loop = (ts) => {
      const y = window.scrollY || window.pageYOffset || 0;
      const drift = Math.sin(ts / 2600) * 130;   // slow ~16s rise and fall, +/-130px of travel
      const p = dustPos(y, drift);
      dustLayers.forEach(el => { el.style.backgroundPosition = p; });
      requestAnimationFrame(loop);
    };
    requestAnimationFrame(loop);
  }
